"cmjornal", "eco.sapo", "jornaldenegocios", "tsf.pt", "sicnoticias", "noticiasaominuto", ] def eh_bloqueada(fonte, link): alvo = (fonte or "").lower() + " " + (link or "").lower() try: dom = --------------------------------------------------------------------------- def _texto_norm(s): return " " + (s or "").lower() + " " def _marcar(texto, dicionario): """Retorna lista de categorias
